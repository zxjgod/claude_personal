## 💎 DesignSync 核心特性

### 项目管理
```typescript
// 创建设计系统项目
DesignSync({ method: "create_project", name: "My Design System" })

// 列出项目
DesignSync({ method: "list_projects" })

// 获取项目信息
DesignSync({ method: "get_project", projectId: "proj-123" })
```

### 文件操作
```typescript
// 列出文件
DesignSync({ method: "list_files", projectId: "proj-123" })

// 读取文件
DesignSync({ method: "get_file", projectId: "proj-123", path: "button.html" })
```

### 安全写入流程
```typescript
// 1. 确定计划（用户审核）
const plan = await DesignSync({
  method: "finalize_plan",
  projectId: "proj-123",
  writes: ["components/**/*.html"],
  deletes: ["old/**/*"]
})

// 2. 写入文件（路径必须在计划中）
await DesignSync({
  method: "write_files",
  planId: plan.planId,
  files: [{ path: "components/button.html", data: "<button>...</button>" }]
})

// 3. 删除文件
await DesignSync({
  method: "delete_files",
  planId: plan.planId,
  paths: ["old/deprecated.html"]
})
```

### 资产注册
```typescript
// 注册预览卡片
DesignSync({
  method: "register_assets",
  planId: plan.planId,
  assets: [{
    name: "Button",
    path: "components/button.html",
    group: "Actions",
    viewport: { width: 400, height: 200 }
  }]
})
```
