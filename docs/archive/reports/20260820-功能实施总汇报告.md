## 🎓 关键技术亮点

### 1. Worktree 隔离自动化
```typescript
if (input.isolation === "worktree") {
  const wt = await createWorktree(workspaceRoot, sessionId);
  try {
    await agentLoop({ workspaceRoot: wt.path, ... });
    await removeWorktree(...);  // 自动清理
  } catch (e) {
    await removeWorktree(...);  // 失败也清理
    throw e;
  }
}
```

### 2. 任务依赖图管理
```typescript
// 双向依赖维护
updateTask(sessionId, task2.id, { addBlockedBy: [task1.id] });
updateTask(sessionId, task1.id, { addBlocks: [task2.id] });

// 查询可执行任务
const available = tasks.filter(t => 
  t.status === "pending" && 
  t.blockedBy.length === 0
);
```

### 3. 缓存感知调度
```typescript
// 智能延迟建议
if (delay >= 295 && delay <= 305) {
  return {
    content: `... ⚠️ Warning: 300s is cache-inefficient.
Use 270s (stay warm) or 1200s+ (amortize miss).`
  };
}
```


## 🚀 实际应用场景

### 场景 1: 并行开发工作流（结构化任务）

```typescript
// 创建任务
TaskCreate({ subject: "Setup DB", description: "..." })
TaskCreate({ subject: "Implement API", description: "..." })
TaskCreate({ subject: "Write tests", description: "..." })

// 设置依赖
TaskUpdate({ taskId: "2", addBlockedBy: ["1"] })
TaskUpdate({ taskId: "3", addBlockedBy: ["2"] })

// 使用 Agent 并行执行（worktree 隔离）
Agent({
  prompt: "Complete task 1",
  isolation: "worktree"
})
```

### 场景 2: 自动化运维（Cron 调度）

```typescript
// 每天凌晨备份
CronCreate({
  cron: "0 2 * * *",
  prompt: "Run backup script",
  durable: true
})

// 每小时健康检查
CronCreate({
  cron: "7 * * * *",
  prompt: "Check system health",
  recurring: true
})
```

### 场景 3: 动态循环监控（ScheduleWakeup）

```typescript
while (true) {
  const status = await checkCI();
  
  if (status === "running") {
    ScheduleWakeup({
      delaySeconds: 270,
      prompt: "Check CI again",
      reason: "Waiting for build (cache warm)"
    });
    break;
  }
  
  if (status === "complete") {
    ScheduleWakeup({ stop: true });
    break;
  }
}
```

### 场景 4: 代码审查流程（ReportFindings）

```typescript
// 审查代码
const issues = await reviewCode({ level: "high" });

// 报告发现
ReportFindings({
  findings: issues.map(i => ({
    file: i.file,
    line: i.line,
    summary: i.message,
    category: i.type,
    failure_scenario: i.scenario,
    verdict: "CONFIRMED"
  })),
  level: "high"
});

// 修复并重新报告
ReportFindings({
  findings: issues.map(i => ({ ...i, outcome: "fixed" })),
  level: "high"
});
```
