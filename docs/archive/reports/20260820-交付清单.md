#### 1. Agent 工具的 Worktree 隔离
```typescript
// 自动创建和清理 worktree
if (input.isolation === "worktree") {
  const wt = await createWorktree(workspaceRoot, sessionId);
  try {
    // 在隔离环境运行
    await agentLoop({ workspaceRoot: wt.path, ... });
    // 成功后清理
    await removeWorktree(workspaceRoot, wt.path, wt.branch);
  } catch (e) {
    // 失败也清理
    await removeWorktree(workspaceRoot, wt.path, wt.branch);
    throw e;
  }
}
```
